  ```python
  import re
  
  def is_palindrome(s: str) -> bool:
      """
      Check if s is a palindrome (alphanumeric chars only, case-insensitive).
  
      Raises TypeError for non-string input.
      Returns False for empty or non-alphanumeric-only strings.
      """
      if not isinstance(s, str):
          raise TypeError("Input must be a string")
      cleaned = re.sub(r'[^a-z0-9]', '', s.lower())
      return len(cleaned) > 0 and cleaned == cleaned[::-1]
  ```
  
  Empty/non-alphanumeric strings now return `False`. Docstring added.
